cmake_minimum_required(VERSION 3.21)
project(ImageSifter VERSION 2.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets Sql Concurrent Test Svg)
include(cmake/LibRaw.cmake)
add_library(sifter_core STATIC src/catalog.cpp src/catalog.h src/formats.cpp src/formats.h)
target_link_libraries(sifter_core PUBLIC Qt6::Core Qt6::Sql sifter_raw)
target_include_directories(sifter_core PUBLIC src)
add_executable(ImageSifter WIN32 src/main.cpp src/window.cpp src/window.h src/photos.cpp src/photos.h src/viewer.cpp src/viewer.h vendor/easyexif/exif.cpp assets/assets.qrc)
target_link_libraries(ImageSifter PRIVATE sifter_core sifter_raw Qt6::Widgets Qt6::Concurrent Qt6::Svg)
target_include_directories(ImageSifter PRIVATE vendor/easyexif)
target_compile_definitions(ImageSifter PRIVATE APP_VERSION="${PROJECT_VERSION}")
if(MSVC)
  target_compile_options(ImageSifter PRIVATE /W4 /utf-8)
  target_compile_options(sifter_core PRIVATE /W4 /utf-8)
else()
  target_compile_options(ImageSifter PRIVATE -Wall -Wextra)
  target_compile_options(sifter_core PRIVATE -Wall -Wextra)
endif()
if(WIN32)
  target_sources(ImageSifter PRIVATE assets/app.rc)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(catalog_tests tests/catalog_tests.cpp)
  target_link_libraries(catalog_tests PRIVATE sifter_core Qt6::Test)
  add_test(NAME catalog COMMAND catalog_tests -o -,txt)
  add_executable(ui_tests tests/ui_tests.cpp src/window.cpp src/photos.cpp src/viewer.cpp vendor/easyexif/exif.cpp assets/assets.qrc)
  target_link_libraries(ui_tests PRIVATE sifter_core sifter_raw Qt6::Widgets Qt6::Concurrent Qt6::Svg Qt6::Test)
  target_include_directories(ui_tests PRIVATE vendor/easyexif)
  target_compile_definitions(ui_tests PRIVATE APP_VERSION="${PROJECT_VERSION}" SIFTER_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME ui COMMAND ui_tests -platform offscreen -o -,txt)
endif()
install(TARGETS ImageSifter RUNTIME DESTINATION .)
